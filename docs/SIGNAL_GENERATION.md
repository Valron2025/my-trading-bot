## 2. Технические индикаторы

### 2.1 RSI (Relative Strength Index)

def calculate_rsi(prices, period=14):
    """
    RSI = 100 - (100 / (1 + RS))
    RS = средний_рост / среднее_падение
    
    Интерпретация:
    - RSI < 30: перепроданность (сигнал к покупке)
    - RSI > 70: перекупленность (сигнал к продаже)
    - RSI 40-60: нейтрально
    """

### 2.2 MACD (Moving Average Convergence Divergence)

def calculate_macd(prices, fast=12, slow=26, signal=9):
    """
    MACD-линия = EMA(12) - EMA(26)
    Сигнальная линия = EMA(MACD-линия, 9)
    Гистограмма = MACD - Сигнальная
    
    Сигналы:
    - MACD пересекает сигнальную снизу вверх → BUY
    - MACD пересекает сигнальную сверху вниз → SELL
    """

### 2.3 Скользящие средние

def moving_averages(prices):
    MA_5 = sum(prices[-5:]) / 5   # Быстрая
    MA_10 = sum(prices[-10:]) / 10
    MA_20 = sum(prices[-20:]) / 20 # Медленная
    
    # Сигналы:
    # MA_5 > MA_10 > MA_20 → восходящий тренд (LONG)
    # MA_5 < MA_10 < MA_20 → нисходящий тренд (SHORT)

### 2.4 Aroon

def calculate_aroon(prices, period=25):
    """
    Aroon Up = ((period - позиция_максимума) / period) * 100
    Aroon Down = ((period - позиция_минимума) / period) * 100
    
    Сигналы:
    - Aroon Up > 70 и Aroon Down < 30 → сильный восходящий тренд
    - Aroon Down > 70 и Aroon Up < 30 → сильный нисходящий тренд
    """

### 3. Формирование сигнала
### 3.1 LONG сигналы (плюсы к score)
Условие	Score	Описание
RSI < 30	+5	Сильная перепроданность
RSI < 35	+3	Перепроданность
RSI < 40	+1	Почти перепроданность

Коррекция в восходящем тренде	+2	Откат для входа
Высокий объём (>1.5x)	+1	Подтверждение

### 3.2 SHORT сигналы (минусы к score)
Условие	Score	Описание
RSI > 70	-5	Сильная перекупленность

Всплеск объёма на перекупленности	-2	Объёмное подтверждение
Откат в нисходящем тренде	-2	Вход на откате

# После расчёта score (от -10 до +10)

if score >= config.long_score_threshold:   # обычно 1 или 2
    action = "BUY"   # LONG
    
elif score <= config.short_score_threshold:  # обычно -1 или -2
    action = "SELL"  # SHORT
    
else:
